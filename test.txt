fun factorial(n) =
{
 if (n < 1)
 {
 output = 1;
 }
 else
 {
 output = n * factorial(n - 1);
 }
}
output;

fun gcd(a, b) =
{
 t = b;
 b = a mod b;
 if (b == 0)
 {
 output = t;
 }
 else
 {
 output = gcd(t, b);
 }
}
output;
